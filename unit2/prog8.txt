
SET SERVEROUTPUT ON;

BEGIN
    FOR R IN
    (
        SELECT ID, NAME, ADDRESS
        FROM CUSTOMER
        WHERE NAME LIKE 'A%'
    )
    LOOP
        DBMS_OUTPUT.PUT_LINE('ID = ' || R.ID);
        DBMS_OUTPUT.PUT_LINE('NAME = ' || R.NAME);
        DBMS_OUTPUT.PUT_LINE('ADDRESS = ' || R.ADDRESS);
    END LOOP;
END;
